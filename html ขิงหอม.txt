    function renderShiftCalendar() {
      const year = currentShiftDate.getFullYear();
      const month = currentShiftDate.getMonth();
      const thaiYear = year + 543;
      
      document.getElementById('shiftCalMonthYear').innerText = `${thaiMonths[month]} พ.ศ. ${thaiYear}`;
      const gridEl = document.getElementById('shiftCalGrid');
      gridEl.innerHTML = '';

      const rawFirstDay = new Date(year, month, 1).getDay();
      const firstDayIndex = (rawFirstDay + 1) % 7; 
      const totalDays = new Date(year, month + 1, 0).getDate();

      for (let i = 0; i < firstDayIndex; i++) {
        gridEl.innerHTML += `<div class="cal-day-cell disabled"></div>`;
      }

      for (let day = 1; day <= totalDays; day++) {
        const formattedMonth = String(month + 1).padStart(2, '0');
        const formattedDay = String(day).padStart(2, '0');
        const dateString = `${year}-${formattedMonth}-${formattedDay}`;

        const currentLoopDate = new Date(year, month, day);
        const jsDay = currentLoopDate.getDay();
        const isWeekend = (jsDay === 6 || jsDay === 0);

        let cellClass = "cal-day-cell";
        let statusBadge = "";

        if (isWeekend) {
          cellClass += " weekend-disabled";
          statusBadge = `<span class="text-[9px] text-gray-400 font-bold">หยุด</span>`;
        } else {
          const key = `${currentLoggedInHospital}_${dateString}`;
          const currentStatus = tempMonthShifts[key] || 'วันนี้ว่าง';

          if (currentStatus === 'วันนี้ว่าง') {
            cellClass += " bg-emerald-50 border-emerald-300 text-emerald-800";
            statusBadge = `<span class="text-[9px] font-bold text-emerald-600">🟢 ว่าง</span>`;
          } else if (currentStatus === 'คิวเต็ม') {
            cellClass += " full-queue";
            statusBadge = `<span class="text-[9px] font-bold text-red-600">🔴 เต็ม</span>`;
          } else {
            cellClass += " bg-amber-50 border-amber-300 text-amber-800";
            statusBadge = `<span class="text-[8px] font-bold truncate text-amber-700">${currentStatus}</span>`;
          }
        }

        const clickAttr = isWeekend ? '' : `onclick="cycleShiftStatus('${dateString}')" ontouchend="cycleShiftStatus('${dateString}')"`;
        gridEl.innerHTML += `
          <div class="${cellClass}" ${clickAttr}>
            <span class="font-bold text-xs">${day}</span>
            ${statusBadge}
          </div>
        `;
      }
    }

    function cycleShiftStatus(dateStr) {
      const key = `${currentLoggedInHospital}_${dateStr}`;
      const currentStatus = tempMonthShifts[key] || 'วันนี้ว่าง';

      if (currentStatus === 'วันนี้ว่าง') {
        tempMonthShifts[key] = 'คิวเต็ม';
        Swal.fire({
          toast: true,
          position: 'top-end',
          icon: 'warning',
          title: `วันที่ ${dateStr} : ตั้งเป็น "คิวเต็ม"`,
          showConfirmButton: false,
          timer: 1200
        });
      } else if (currentStatus === 'คิวเต็ม') {
        Swal.fire({
          title: 'ระบุช่วงเวลาให้บริการ',
          input: 'text',
          inputPlaceholder: 'เช่น 09:00 - 12:00 น.',
          showCancelButton: true,
          confirmButtonText: 'ตกลง',
          cancelButtonText: 'ข้าม'
        }).then((res) => {
          if (res.isConfirmed && res.value) {
            tempMonthShifts[key] = res.value;
          } else {
            tempMonthShifts[key] = 'วันนี้ว่าง';
          }
          renderShiftCalendar();
        });
        return;
      } else {
        tempMonthShifts[key] = 'วันนี้ว่าง';
      }
      renderShiftCalendar();
    }

    function renderCalendarUI(mode) {
      const isCheck = (mode === 'check');
      const dateObj = isCheck ? currentCheckDate : currentBookDate;
      const year = dateObj.getFullYear();
      const month = dateObj.getMonth();

      const thaiYear = year + 543;
      const monthYearEl = document.getElementById(isCheck ? 'checkCalMonthYear' : 'bookCalMonthYear');
      if (monthYearEl) {
        monthYearEl.innerText = `${thaiMonths[month]} พ.ศ. ${thaiYear}`;
      }

      const gridEl = document.getElementById(isCheck ? 'checkCalGrid' : 'bookCalGrid');
      if (!gridEl) return;
      gridEl.innerHTML = '';

      const rawFirstDay = new Date(year, month, 1).getDay();
      const firstDayIndex = (rawFirstDay + 1) % 7; 
      const totalDays = new Date(year, month + 1, 0).getDate();

      for (let i = 0; i < firstDayIndex; i++) {
        gridEl.innerHTML += `<div class="cal-day-cell disabled"></div>`;
      }

      const selectedHosp = isCheck ? document.getElementById('checkQueueHospital').value : document.getElementById('hospitalSelect').value;

      for (let day = 1; day <= totalDays; day++) {
        const formattedMonth = String(month + 1).padStart(2, '0');
        const formattedDay = String(day).padStart(2, '0');
        const dateString = `${year}-${formattedMonth}-${formattedDay}`;

        const currentLoopDate = new Date(year, month, day);
        const jsDay = currentLoopDate.getDay();
        const isWeekend = (jsDay === 6 || jsDay === 0);

        let cellClass = "cal-day-cell";
        let statusBadge = "";

        if (isWeekend) {
          cellClass += " weekend-disabled";
          statusBadge = `<span class="text-[9px] text-gray-400 font-bold">หยุด</span>`;
        } else if (selectedHosp) {
          const shiftKey = `${selectedHosp}_${dateString}`;
          const shiftStatus = shiftSchedules[shiftKey] || 'วันนี้ว่าง';

          if (shiftStatus === 'คิวเต็ม') {
            cellClass += " full-queue";
            statusBadge = `<span class="text-[9px] text-red-600 font-bold">🔴 เต็ม</span>`;
          } else if (shiftStatus === 'วันนี้ว่าง') {
            statusBadge = `<span class="text-[9px] text-emerald-600 font-bold">🟢 ว่าง</span>`;
          } else {
            statusBadge = `<span class="text-[8px] text-amber-700 font-bold truncate">${shiftStatus}</span>`;
          }
        }

        if (!isCheck && selectedBookingDateStr === dateString) {
          cellClass += " selected";
        }

        const clickAction = isWeekend ? '' : `onclick="onCalendarCellClick('${mode}', '${dateString}', ${day}, ${isWeekend})" ontouchend="onCalendarCellClick('${mode}', '${dateString}', ${day}, ${isWeekend})"`;

        gridEl.innerHTML += `
          <div id="${mode}-day-${day}" class="${cellClass}" ${clickAction}>
            <span class="font-bold text-xs">${day}</span>
            ${statusBadge}
          </div>
        `;
      }

      if (isCheck && selectedHosp) {
        renderQueueListForSelectedHospital(selectedHosp);
      }
    }

        }

        const shiftKey = `${hosp}_${dateStr}`;
        const shiftStatus = shiftSchedules[shiftKey] || 'วันนี้ว่าง';

          Swal.fire('ขออภัย คิวเต็มแล้ว', `วันที่ ${dateStr} ของ ${hosp} มีสถานะ "คิวเต็ม" ไม่สามารถจองได้ค่ะ`, 'error');

    function renderQueueListForSelectedHospital(hospName, specificDateStr = "") {
      const container = document.getElementById('queueResultContainer');
      let bookings = allBookingsCache.filter(b => b.hospital === hospName && b.status !== 'ยกเลิกแล้ว');

      if (specificDateStr) {
        bookings = bookings.filter(b => b.date === specificDateStr);
      }

      let shiftInfo = "";
      if (specificDateStr) {
        const shiftKey = `${hospName}_${specificDateStr}`;
        const shiftVal = shiftSchedules[shiftKey] || 'วันนี้ว่าง';
        shiftInfo = `<div class="bg-matcha-light p-2.5 rounded-xl text-xs text-matcha-dark font-bold mb-2">📌 สถานะเวรวันที่ ${specificDateStr}: ${shiftVal}</div>`;
      }

      if (bookings.length === 0) {
        container.innerHTML = shiftInfo + `<p class="text-center text-xs text-gray-400 py-4">ไม่พบรายการนัดหมายในวันที่เลือก</p>`;
        return;
      }

      let html = shiftInfo + `<p class="text-xs font-bold text-matcha-dark mt-2">📋 รายการนัดหมาย (${bookings.length} คิว):</p>`;
      bookings.forEach(b => {
        html += `
          <div class="bg-gray-50 p-3 rounded-2xl border border-gray-100 text-xs space-y-1">
            <div class="flex justify-between items-center">
              <span class="font-bold text-gray-700">${b.refNo} (${b.date})</span>
              <span class="px-2 py-0.5 bg-matcha-light text-matcha-dark rounded-full font-bold">${b.status}</span>
            </div>
            <p class="text-gray-600">👤 ${b.fullName} | 🕒 ${b.startTime} น.</p>
            <p class="text-gray-500">💆🏻‍♀️ ${b.service} (${b.duration || '1 ชม.'}) - ${b.therapist}</p>
          </div>
        `;
      });
      container.innerHTML = html;
    }

    function calculatePrice() {
      const service = document.getElementById('serviceSelect').value;
      const duration = document.getElementById('durationSelect').value;
      const rights = document.getElementById('rightsSelect').value;

      let basePrice = 0;
      let extraFee = 0;

      if (service && priceMatrix[service] && priceMatrix[service][rights]) {
        const rates = priceMatrix[service][rights];
        extraFee = rates[duration] !== undefined ? rates[duration] : (rates["1 ชั่วโมง"] || 0);

        if (service === "5. ทับหม้อเกลือ" && rates["ตามสิทธิ์"]) {
          basePrice = rates["ตามสิทธิ์"];
        }
      }

      const totalPrice = basePrice + extraFee;

      document.getElementById('basePriceDisplay').innerText = `${basePrice} บาท`;
      document.getElementById('extraFeeDisplay').innerText = `${extraFee} บาท`;
      document.getElementById('totalPriceDisplay').innerText = `${totalPrice} บาท`;

      return { basePrice, extraFee, totalPrice };
    }

    function loadAdminData() {
      google.script.run.withSuccessHandler((data) => {
        allBookingsCache = data;
        updateDashboardView();
      }).getBookings();
    }

    function updateDashboardView() {
      const hospFilter = document.getElementById('dashHospitalFilter').value;
      const timeFilter = document.getElementById('dashTimeFilter').value;

      let filtered = [...allBookingsCache];
      if (hospFilter !== "ALL") filtered = filtered.filter(b => b.hospital === hospFilter);

      const validBookings = filtered.filter(b => b.status !== 'ยกเลิกแล้ว');
      document.getElementById('dashKpiTotal').innerText = validBookings.length;
      document.getElementById('dashKpiIncome').innerText = validBookings.reduce((sum, b) => sum + (Number(b.price) || 0), 0).toLocaleString() + ' ฿';
      document.getElementById('dashKpiGenderRatio').innerText = `${validBookings.filter(b=>b.gender==='ชาย').length} : ${validBookings.filter(b=>b.gender==='หญิง').length}`;
      document.getElementById('dashKpiTopService').innerText = "นวดบำบัด";

      renderAdminBookingList(filtered);
    }

    function renderAdminBookingList(data) {
      const container = document.getElementById('adminBookingList');
      container.innerHTML = data.length === 0 ? '<p class="text-center text-xs text-gray-400 py-4">ไม่มีรายการข้อมูล</p>' : '';
      data.forEach(b => {
        container.innerHTML += `
          <div class="bg-white p-3 rounded-2xl shadow-sm border border-gray-100 text-xs space-y-1">
            <div class="flex justify-between items-center">
              <span class="font-bold text-gray-700">${b.refNo}</span>
              <span class="px-2 py-0.5 rounded-full font-bold text-green-600 bg-green-50">${b.status}</span>
            </div>
            <p class="text-gray-600">👤 ${b.fullName} | 📞 ${b.phone}</p>
            <p class="text-gray-500">🏥 ${b.hospital} | 📅 ${b.date} (${b.startTime} น.) - ${b.service}</p>
          </div>
        `;
      });
    }
